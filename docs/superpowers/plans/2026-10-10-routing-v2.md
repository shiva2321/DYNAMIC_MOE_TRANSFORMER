# Routing v2 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make PhyloMoE (a) train each expert on batch-consistent assignments so overlapping regimes stop contaminating one another, and (b) route through a cheap fingerprint shortlist whose novelty decisions are provably identical to the exact gate. Each is a flag, is pre-registered, and becomes the default only if it passes.

**Architecture:** Stage 1 adds one pure method (`_batch_consistent`) to `PhyloMoE.observe` (batch vote). Stage 2 adds `growmoe/fingerprint.py` and a two-stage gate inside `PhyloMoE._gate_matrices` (k-means sub-centroid shortlist, exact conformal scores on the shortlist, an exact "second look" for every sample the shortlist would call novel), after first routing `observe` through `_gate_matrices`. Stage 3 is a conditional throwaway spike. Experiments reuse the M0/M2-A harness (`run_m0.py` parallel runner and `freeze_check`, `fairfight.run_fast / tune_steps / run_record`, `metrics.paired_diff`).

**Tech Stack:** Python 3, PyTorch (CPU), NumPy, pytest. Tests: `python -m pytest tests/growmoe -q` (baseline before this plan: 432 passed, about 140 s).

**Spec:** `docs/superpowers/specs/2026-10-10-routing-v2-design.md` (executors read both; "R3" means section 3 of `research/growmoe_results/rv2/RESULTS_RV2_SPIKES.md`, the script-generated spike evidence).

## Review Focus

Failure modes the spec implies that are most likely to bite a person using this software (each is pinned by a test in the owning task):

1. A tiny batch (a single sample, or fewer than `batch_vote_min` accepted) with either feature on must not crash or move anything (Tasks 2 and 9).
2. Unlabeled batches (`y=None`) with the vote on must work (Task 2).
3. The `no_conformal` ablation with both features on: the vote still applies, the cascade stays off (Tasks 2 and 9).
4. A batch in which every sample is novel (nothing accepted; the cascade's second look must cover every row) (Task 9).
5. A batch that mixes two known regimes: the vote moves the minority into the majority's expert by design; the behaviour is documented by a unit test and bounded by the G-MIX guard (Task 2).
6. A pool whose experts are all infants (always scored exactly) and repeated runs (determinism) (Task 9).

```python
"""RV2-A: overlapping regimes and a mixed tail for the function-regime stream (spec section 4).

``overlap`` gives the regimes' centres a shared component so their pairwise cosine is about ``overlap``; at 0 the stream is bit-identical to
the pilot. ``mixed_tail`` appends 70/30 segments of regimes the pool already knows, built from their unused second halves.

import pytest
import torch

from growmoe.streams import _balanced_teacher, _long, _rng, _tensor, _unit
from growmoe.streams_m1 import function_regimes


def _pilot(seed, n_regimes, n_train, n_test, d=128, n_classes=8):
    """function_regimes exactly as it was before ``overlap`` existed."""
    rng = _rng(seed, 0xF5)
    centres = _unit(rng.standard_normal((n_regimes, d)))
    pools, tests = {}, {}
    for r in range(n_regimes):
        teacher = _balanced_teacher(8, 1, d, n_classes, seed=seed * 1000 + r)

        def draw(n):
            z = rng.standard_normal((n, d))
            y = teacher(_tensor(_unit(z))).argmax(1).detach().numpy()
            return _unit(8.0 * centres[r] + z), y

        pools[r] = draw(2 * n_train)
        tests[r] = draw(n_test)
    segs, visits = [], {}
    for r in list(range(n_regimes)) + [0, n_regimes // 2]:
        v = visits.get(r, 0)
        visits[r] = v + 1
        sl = slice(v * n_train, (v + 1) * n_train)
        segs.append((_tensor(pools[r][0][sl]), _long(pools[r][1][sl])))
    return segs, {r: (_tensor(x), _long(y)) for r, (x, y) in tests.items()}


def _mean_pair_cosine(st):
    c = torch.as_tensor(st.meta["centres"], dtype=torch.float32)
    g = c @ c.T
    n = len(c)
    return float((g.sum() - g.diag().sum()) / (n * (n - 1)))


def test_overlap_zero_is_bit_identical_to_the_pilot_stream():
    st = function_regimes(5, n_regimes=4, n_train=64, n_test=32, overlap=0.0)
    segs, tests = _pilot(5, 4, 64, 32)
    assert len(st.segments) == len(segs)
    for seg, (x, y) in zip(st.segments, segs):
        assert torch.equal(seg.x, x) and torch.equal(seg.y, y)
    for r, (x, y) in tests.items():
        assert torch.equal(st.regimes[r].x_te, x) and torch.equal(st.regimes[r].y_te, y)
```

```python
def function_regimes(seed: int = 0, n_regimes: int = 15, d: int = 128, n_classes: int = 8, n_train: int = 800, n_test: int = 400,
                     centre_scale: float = 8.0, teacher_width: int = 8, teacher_depth: int = 1,
                     overlap: float = 0.0, mixed_tail: int = 0, minor_share: float = 0.3) -> Stream:
    """``overlap`` is the target cosine between regime centres (0 = the pilot stream, bit for bit). ``mixed_tail`` appends that many segments
    in which a regime the pool already knows arrives mixed with a second known one (``1 - minor_share`` / ``minor_share``), built from the
    unused second halves of regimes that no revisit uses; ``meta["mixed_pairs"]`` and ``meta["minority_regimes"]`` name them."""
    if n_regimes < 2:
        raise ValueError("a function-regime stream needs at least 2 regimes")
    if not 0.0 <= overlap < 1.0:
        raise ValueError("overlap (the target cosine between regime centres) must lie in [0, 1)")
    if not 0.0 < minor_share <= 0.5:
        raise ValueError("minor_share must lie in (0, 0.5]")
    free = [r for r in range(n_regimes) if r not in (0, n_regimes // 2)]                 # regimes whose second half no revisit uses
    pairs = [(free[i], free[i + 1]) for i in range(0, len(free) - 1, 2)]
    if not 0 <= mixed_tail <= len(pairs):
        raise ValueError(f"mixed_tail must lie in [0, {len(pairs)}] for {n_regimes} regimes")
    rng = _rng(seed, 0xF5)
    g = rng.standard_normal((n_regimes, d))
    if overlap == 0.0:
        centres = _unit(g)
    else:                                                                                 # the shared direction has its own generator: the main stream is untouched
        shared = _rng(seed, 0xF6).standard_normal((1, d))
        centres = _unit(np.sqrt(1.0 - overlap) * _unit(g) + np.sqrt(overlap) * _unit(shared))
    regimes, pools = {}, {}
    for r in range(n_regimes):
        teacher = _balanced_teacher(teacher_width, teacher_depth, d, n_classes, seed=seed * 1000 + r)

        def draw(n: int):
            z = rng.standard_normal((n, d))
            y = teacher(_tensor(_unit(z))).argmax(1).detach().numpy()
            return _unit(centre_scale * centres[r] + z), y

        x_tr, y_tr = draw(2 * n_train)                                                  # two chunks: first visit and one revisit
        x_te, y_te = draw(n_test)
        regimes[r] = Regime(r, f"fn{r}", _tensor(x_te), _long(y_te), _long(np.arange(n_test)), meta={"type": "function"})
        pools[r] = (x_tr, y_tr)
    schedule = list(range(n_regimes)) + [0, n_regimes // 2]
    segments, visits = [], {}
    for r in schedule:
        v = visits.get(r, 0)
        visits[r] = v + 1
        sl = slice(v * n_train, (v + 1) * n_train)
        segments.append(_segment(r, v == 0, _tensor(pools[r][0][sl]), _long(pools[r][1][sl]), np.arange(sl.start, sl.stop)))
    n_minor = int(round(minor_share * n_train))
    n_major = n_train - n_minor
    for a, b in pairs[:mixed_tail]:
        xa, ya = pools[a][0][n_train:n_train + n_major], pools[a][1][n_train:n_train + n_major]
        xb, yb = pools[b][0][n_train:n_train + n_minor], pools[b][1][n_train:n_train + n_minor]
        segments.append(_segment(a, False, _tensor(np.concatenate([xa, xb])), _long(np.concatenate([ya, yb])), np.arange(n_train, 2 * n_train)))
        schedule.append(a)
    meta = {"schedule": schedule, "centres": centres, "overlap": overlap, "mixed_pairs": pairs[:mixed_tail],
            "minority_regimes": [b for _, b in pairs[:mixed_tail]]}
    return Stream("function_regimes", d, n_classes, regimes, segments, meta)
```

```python
"""RV2-A stage 1: the batch vote (spec section 4).

Per-sample routing lets a fraction of every batch train the wrong expert when regimes overlap (training purity 78 % at centre cosine 0.3, R5).
When one expert holds at least ``batch_vote_share`` of a batch's accepted samples, all of them train that expert.
"""

import numpy as np
import pytest
import torch

from growmoe.config import PhyloCfg
from growmoe.pools.phylo import PhyloMoE
from growmoe.streams_m1 import function_regimes

A_KW = dict(q=0.02, alpha_spawn=0.01, m_min=32, route_rule="accept_then_logf")
SEEDS = (100, 101, 102, 103)


def _vote(assigned, **kw):
    pool = PhyloMoE(PhyloCfg(d=8, n_classes=4, batch_vote=True, **kw))
    return pool._batch_consistent(torch.tensor(assigned))


def _pool(st, steps=5, seed=100, **kw):
    return PhyloMoE(PhyloCfg(seed=seed, d=st.d, n_classes=st.n_classes, inner_steps=steps, **A_KW, **kw))


def _purity(st, pool, seed):
    """Share of the samples each expert trained on that came from its dominant regime (summed over experts)."""
    counts = {}
    for (_, seg, _), (x, y, _) in zip(st._walk(64, seed), st.algo_batches(64, seed)):
        for e in pool.observe(x, y).assigned.tolist():
            if e >= 0:
                counts[(seg.regime, e)] = counts.get((seg.regime, e), 0) + 1
    per_expert = {}
    for (_, e), c in counts.items():
        per_expert.setdefault(e, []).append(c)
    return sum(max(v) for v in per_expert.values()) / max(1, sum(counts.values()))


def _purity_for(seed, vote, overlap):
    st = function_regimes(seed, n_regimes=3, n_train=256, n_test=100, overlap=overlap)
    return _purity(st, _pool(st, seed=seed, batch_vote=vote), seed)
```

```python
# ------------------------------------------------------------------------------------------------ RV2-A verdicts (stage 1: batch vote)
# research/growmoe_results/rv2a/PREREG.md: does letting a batch train one expert fix training contamination under overlap without hurting anything else?
BARS_RV2A = {"V1": 8.0, "V2": 1.0, "V3": 3.0}
EXPECTED_RV2A = {k: "PASS" for k in ("V1", "V2", "V3", "G1", "G2", "G3", "G4", "G5", "G6", "G7", "GM")}
_GUARDS_RV2A = {"G1": "S1", "G2": "S2-5", "G3": "S2-30", "G4": "S4", "G5": "OV-0.0", "G6": "OV-0.3-MIXED"}
_MIX_STREAMS = ("MIX-0.0", "MIX-0.3")
_GUARD_IDS_RV2A = tuple(_GUARDS_RV2A) + ("G7", "GM")


def _rules_rv2a(bars: dict, cand: str, ref: str) -> dict:
    guard = f"{cand} - {ref} >= -3.0 points and the 95% interval is not entirely below 0"
    rules = {
        "V1": f"OV-0.3: mean paired {cand} - {ref} >= +{bars['V1']:.1f} points and the 95% interval excludes 0",
        "V2": f"F-15 and F-30: mean paired {cand} - {ref} >= +{bars['V2']:.1f} points and the 95% interval excludes 0",
        "V3": f"OV-0.6: mean paired {cand} - {ref} >= +{bars['V3']:.1f} points",
        "G7": f"S7: {cand} raises at least 3 alarms per run on average (the alarm still catches a label remap)",
        "GM": f"MIX-0.0 and MIX-0.3, accuracy on the minority regimes: {guard}",
    }
    rules.update({gid: f"{stream}: {guard}" for gid, stream in _GUARDS_RV2A.items()})
    return rules


def _guard_points(ix: dict, stream: str, cand: str, ref: str):
    d = _paired_points(ix, (stream, cand), (stream, ref), -3.0)
    if d is not None:                                                                   # a regression that is clearly below 0 fails even if small
        d["holds"] = bool(d["holds"] and not (d["ci_hi"] is not None and d["ci_hi"] < 0.0))
    return d


def rv2a_verdicts(runs: Sequence[dict], bars: dict = None, expected: dict = None, candidate: str = "A_vote", reference: str = "A_new") -> dict:
    """The RV2-A statements on per-seed run records. Multi-stream statements follow the M0 rule (PASS on all, FAIL on none, PARTIAL otherwise,
    NOT MEASURABLE when data are missing). ``bars`` / ``expected`` come from the pre-registration (bars_rv2a.json)."""
    bars = {**BARS_RV2A, **(bars or {})}
    exp = {**EXPECTED_RV2A, **(expected or {})}
    ix = _index(runs)
    detail = {
        "V1": _paired_points(ix, ("OV-0.3", candidate), ("OV-0.3", reference), bars["V1"], need_interval=True),
        "V2": {s: _paired_points(ix, (s, candidate), (s, reference), bars["V2"], need_interval=True) for s in ("F-15", "F-30")},
        "V3": _paired_points(ix, ("OV-0.6", candidate), ("OV-0.6", reference), bars["V3"]),
    }
    for gid, stream in _GUARDS_RV2A.items():
        detail[gid] = _guard_points(ix, stream, candidate, reference)
    detail["GM"] = {s: _guard_points(ix, s, candidate, reference) for s in _MIX_STREAMS}
    verdict = {k: _combine(v) for k, v in detail.items()}
    s7 = ix.get(("S7", candidate))
    if not s7:
        detail["G7"], verdict["G7"] = None, "NOT MEASURABLE"
    else:
        mean_alarms = float(np.mean([r.get("event_counts", {}).get("alarm", 0) for r in s7.values()]))
        detail["G7"] = {"n": len(s7), "mean_alarms": mean_alarms}
        verdict["G7"] = "PASS" if mean_alarms >= 3.0 else "FAIL"
    rules = _rules_rv2a(bars, candidate, reference)
    return {k: {"rule": rules[k], "verdict": verdict[k], "detail": detail[k], "expected": exp[k],
                "as_expected": None if verdict[k] == "NOT MEASURABLE" else verdict[k] == exp[k]} for k in EXPECTED_RV2A}


def _clears_rv2a(result: dict):
    return result["V1"]["verdict"], [g for g in _GUARD_IDS_RV2A if result[g]["verdict"] != "PASS"]


def rv2a_decision(plain: dict, restricted: dict = None) -> str:
    """What the pre-registration says to do, given the plain-vote verdicts and (if the fallback ran) the restricted-vote verdicts."""
    v1, bad = _clears_rv2a(plain)
    if v1 == "NOT MEASURABLE":
        return "The RV2-A decision cannot be applied: V1 was not measurable."
    if v1 != "PASS":
        return (f"V1 {v1}: the training-contamination mechanism is not supported at the pre-registered margin. Stop: keep batch_vote off, "
                "re-diagnose, and do not start stage 3 on this premise.")
    if not bad:
        return ("V1 PASS and every guard holds: adopt the plain vote (batch_vote=True) as the PhyloMoE default, test-first, with the historical "
                "entry points pinned.")
    if restricted is None:
        return (f"V1 PASS but guard(s) {', '.join(bad)} did not pass: run the restricted vote (a sample moves only if the modal expert's own gate "
                "accepts it) as the pre-declared fallback on the same seeds.")
    r1, rbad = _clears_rv2a(restricted)
    if r1 == "PASS" and not rbad:
        return "The plain vote failed a guard and the restricted vote passes V1 and every guard: adopt the restricted vote, test-first, with the historical entry points pinned."
    return ("Neither the plain nor the restricted vote passes V1 and every guard: adopt neither; batch_vote stays off and the cause goes into "
            "the stage-3 design.")
```

```python
def mix_pairs(st: Stream) -> Stream:
    """Every two consecutive segments interleaved: each batch is a 50/50 mixture of two regimes (a guard stream, kept out of the package)."""
    segs, out, i = st.segments, [], 0

        kw["batch_vote_restricted"] = True
    return lambda: PhyloMoE(PhyloCfg(seed=seed, d=st.d, n_classes=st.n_classes, inner_steps=P["steps"], **kw))

    return tm_new["A_new"][f]["steps"] if method == "A_new" else tm_vote[method][f]["steps"]

def tune_task(spec: dict) -> dict:
    fam, smoke, arm = spec["family"], spec["smoke"], spec["arm"]
    seeds = M.SMOKE_DEV if smoke else M.DEV_SEEDS
    assert all(s >= 100 for s in seeds), "tuning must use dev seeds only"
    name = M.TUNE_STREAM[fam]
    streams = {s: get_stream(name, s, smoke) for s in seeds}

    def score(steps: int) -> float:
        return float(np.mean([run_fast(factory(arm, streams[s], s, {"steps": steps}), streams[s], s, eval_every="final").acc for s in seeds]))

    t0 = time.perf_counter()
    res = tune_steps(score, grid=M.SMOKE_GRID if smoke else M.STEP_GRID, factor=M.STEP_FACTOR, cap=M.STEP_CAP)
    return {"family": fam, "arm": arm, "steps": res.best, "at_edge": res.at_edge, "table": {str(k): v for k, v in res.table.items()},
            "stream": name, "dev_seeds": list(seeds), "seconds": time.perf_counter() - t0, "peak_mb": M._peak_mb()}


def cmd_tune(args) -> None:
    out = out_dir(args)
    specs = [dict(family=f, smoke=args.smoke, arm=args.arm) for f in ("func", "perm", "types")]
    results = M._parallel(tune_task, specs, lambda s: out / "tune" / f"{s['arm']}__{s['family']}.json", args.workers, args.threads, "tune", args.force,
                          tag_of=lambda s: s["family"])

    tuning = json.loads(path.read_text(encoding="utf-8")) if path.exists() else {}
    tuning[args.arm] = {r["family"]: {"steps": r["steps"], "at_edge": r["at_edge"], "table": r["table"], "stream": r["stream"], "dev_seeds": r["dev_seeds"]}
                        for r in results}
    path.write_text(json.dumps(tuning, indent=1), encoding="utf-8")
    for f, v in tuning[args.arm].items():
        print(f"  {args.arm} {f:6s} steps={v['steps']:>3d}" + ("   ** at the edge of the search **" if v["at_edge"] else ""))


# ------------------------------------------------------------------------------------------------ runs (dev and confirmatory share the task)
def run_task(spec: dict) -> dict:
    name, seed, smoke = spec["stream"], spec["seed"], spec["smoke"]
    tm_new = json.loads(Path(spec["tuning_m2a"]).read_text(encoding="utf-8"))
    tm_vote = json.loads(Path(spec["tuning_rv2a"]).read_text(encoding="utf-8"))
    st = get_stream(name, seed, smoke)
    t0, recs = time.perf_counter(), []
    for m in spec["arms"]:
        P = {"steps": steps_for(m, name, tm_new, tm_vote)}
        res = run_fast(factory(m, st, seed, P), st, seed, eval_every="final")
        rec = run_record(name, m, seed, res, **P)
        minority = st.meta.get("minority_regimes") or []
        if name.startswith("MIX-") and minority:                                        # the statement reads the minority regimes only
            rec["acc_all"] = rec["acc"]
            rec["acc"] = float(np.mean([res.per_regime[r] for r in minority]))
        recs.append(rec)
    return {"spec": {k: spec[k] for k in ("stream", "seed", "smoke", "arms")}, "records": recs, "seconds": time.perf_counter() - t0, "peak_mb": M._peak_mb()}


def _specs(args, out: Path, seeds, arms, streams):

                   for n in streams for s in seeds], key=lambda s: (order[s["stream"]], s["seed"]))

def _floor_half(x: float) -> float:
    return math.floor(x * 2.0) / 2.0


def make_bars(effects: dict):
    """Bars stay at the spec's value unless the dev effect is below 1.5 x the bar; then the bar drops to effect / 1.5 (at least 0.5)."""
    members = {"V1": ["OV-0.3"], "V2": ["F-15", "F-30"], "V3": ["OV-0.6"]}
    bars, expected = {}, {}
    for sid, streams in members.items():
        eff = min(effects[s] for s in streams)
        bar = BARS_RV2A[sid]
        if eff < 1.5 * bar:
            bar = max(0.5, _floor_half(max(eff, 0.0) / 1.5))
        bars[sid] = bar
        expected[sid] = "PASS" if eff > bar else "FAIL"
    for gid, stream in {"G1": "S1", "G2": "S2-5", "G3": "S2-30", "G4": "S4", "G5": "OV-0.0", "G6": "OV-0.3-MIXED"}.items():
        expected[gid] = "FAIL" if effects[stream] < -3.0 else "PASS"
    expected["GM"] = "FAIL" if min(effects["MIX-0.0"], effects["MIX-0.3"]) < -3.0 else "PASS"
    expected["G7"] = "PASS"
    return bars, expected


def _sha(path: Path) -> str:

def _pts(d) -> str:
    if d is None:
        return "missing"
    lo = "-inf" if d["ci_lo"] is None else f"{d['ci_lo']:+.1f}"
    hi = "+inf" if d["ci_hi"] is None else f"{d['ci_hi']:+.1f}"
    return f"{d['mean_diff_points']:+.1f} pts [{lo}, {hi}] (n={d['n']})"


def _numbers(k: str, v: dict) -> str:
    d = v["detail"]
    if v["verdict"] == "NOT MEASURABLE" and (d is None or (isinstance(d, dict) and all(x is None for x in d.values()))):
        return "missing data"
    if k == "G7":
        return f"mean alarms per run {d['mean_alarms']:.2f} (n={d['n']})"
    if k in ("V2", "GM"):
        return "; ".join(f"{s}: {_pts(x)}" for s, x in d.items())
    return _pts(d)


def cmd_report(args) -> None:
    out = out_dir(args)
    tasks = [json.loads(p.read_text(encoding="utf-8")) for p in sorted((out / "runs").glob("*.json"))]
    runs = [r for t in tasks for r in t["records"]]
    bp = out / "bars_rv2a.json"
    doc = json.loads(bp.read_text(encoding="utf-8")) if bp.exists() else {"bars": {}, "expected": {}}
    plain = rv2a_verdicts(runs, doc["bars"], doc["expected"], candidate="A_vote")
    restricted = rv2a_verdicts(runs, doc["bars"], doc["expected"], candidate="A_voter") if any(r["method"] == "A_voter" for r in runs) else None
    decision = rv2a_decision(plain, restricted)
    (out / "rv2a_runs.json").write_text(json.dumps({"runs": runs, "bars": doc, "verdicts": plain, "restricted": restricted, "decision": decision}, indent=1), encoding="utf-8")
    by: dict = {}
    for r in runs:
        by.setdefault((r["stream"], r["method"]), []).append(r)
    seeds = sorted({r["seed"] for r in runs})
    L = ["# RV2-A results (generated by run_rv2a.py; do not edit by hand)\n"]
    if args.smoke:
        L.append("> **SMOKE RUN. Tiny streams and grids. These numbers mean nothing.**\n")
    L.append(f"* PREREG.md sha256 {_sha(out / 'PREREG.md')}; bars_rv2a.json {_sha(bp)}; tuning_rv2a.json {_sha(out / 'tuning_rv2a.json')}; this script {_sha(Path(__file__))}; "
             f"fairfight.py {_sha(ROOT / 'growmoe/fairfight.py')}; phylo.py {_sha(ROOT / 'growmoe/pools/phylo.py')}.")
    if seeds:
        L.append(f"* Seeds {seeds[0]}-{seeds[-1]} ({len(seeds)}); compute summed over tasks {sum(t['seconds'] for t in tasks) / 60:.0f} min; "
                 f"largest worker peak memory {max(t['peak_mb'] for t in tasks):.0f} MB.\n")
    for title, res in (("Verdicts: plain vote (A_vote - A_new)", plain), ("Verdicts: restricted vote (A_voter - A_new)", restricted)):
        if res is None:
            continue
        L += [f"## {title}\n", "| ID | Rule | Verdict | Expected | Numbers |", "|---|---|---|---|---|"]
        for k, v in res.items():
            mark = {True: "as expected", False: "**AGAINST expectation**", None: "-"}[v["as_expected"]]
            L.append(f"| {k} | {v['rule']} | **{v['verdict']}** | {v['expected']} ({mark}) | {_numbers(k, v)} |")
        L.append("")
    L += [f"**Decision.** {decision}\n", "## Results per stream (final accuracy %; mean [95% t interval] over seeds; MIX-* = minority regimes)\n"]
    for name in STREAM_ORDER:
        present = [m for m in LABEL if (name, m) in by]
        if not present:
            continue
        L += [f"### {name}\n", "| Arm | Accuracy % | Experts | Spawns / run | Extends / run | Alarms / run | n |", "|---|---|---|---|---|---|---|"]
        for m in present:
            rows = by[(name, m)]
            ev = {k: float(np.mean([r["event_counts"].get(k, 0) for r in rows])) for k in ("spawn", "extend", "alarm")}
            L.append(f"| {LABEL[m]} | {M._fmt_ci([r['acc'] for r in rows])} | {M._mean(rows, 'n_experts'):.1f} | {ev['spawn']:.1f} | {ev['extend']:.1f} | {ev['alarm']:.1f} | {len(rows)} |")
        for cand in ("A_vote", "A_voter"):
            ra, rb = {r["seed"]: r for r in by.get((name, cand), [])}, {r["seed"]: r for r in by.get((name, "A_new"), [])}
            common = sorted(set(ra) & set(rb))
            if len(common) >= 2:
                d = paired_diff([100 * ra[s]["acc"] for s in common], [100 * rb[s]["acc"] for s in common])
                L.append(f"\n{cand} - A_new: {d['mean_diff']:+.1f} pts [{d['ci_lo']:+.1f}, {d['ci_hi']:+.1f}], p(paired t) = {d['p_t']:.3g}, n = {d['n']}")
        L.append("")
    tp = out / "tuning_rv2a.json"
    if tp.exists():
        tm = json.loads(tp.read_text(encoding="utf-8"))
        L += ["## Training budgets (steps per batch; dev score by steps)\n", "| Arm | Family | Dev stream | Steps | At the edge? | Dev score by steps (%) |", "|---|---|---|---|---|---|"]
        for arm, fams in tm.items():
            for f, v in fams.items():
                L.append(f"| {arm} | {f} | {v['stream']} | {v['steps']} | {'yes' if v['at_edge'] else 'no'} | "
                         + ", ".join(f"{k}: {100 * s:.1f}" for k, s in sorted(v["table"].items(), key=lambda kv: int(kv[0]))) + " |")
    (out / "RESULTS_RV2A.md").write_text("\n".join(L) + "\n", encoding="utf-8")
    print(f"wrote {out / 'RESULTS_RV2A.md'}")
    for k, v in plain.items():
        print(f"  {k}: {v['verdict']:15s} (expected {v['expected']})")
    print(" ", decision)
```

```python
"""Stage-1 fingerprints for the gate shortlist (docs/superpowers/specs/2026-10-10-routing-v2-design.md section 5).

An expert's fingerprint is k unit sub-centroids of its stored prototypes. A first, cheap pass scores an input against every expert's
sub-centroids (best cosine) and keeps the top few experts; only those get the exact conformal score. No pool state lives here.
"""

import torch
import torch.nn.functional as F
from torch import Tensor


def spherical_kmeans(points: Tensor, k: int, iters: int = 10, seed: int = 0) -> Tensor:
    """k unit-length centroids of unit-length ``points`` (rows); k is capped at the number of points."""
    n = points.shape[0]
    k = max(1, min(int(k), n))
    g = torch.Generator().manual_seed(int(seed))
    centroids = points[torch.randperm(n, generator=g)[:k]].clone()
    for _ in range(iters):
        assign = (points @ centroids.T).argmax(1)
        for j in range(k):
            members = assign == j
            if bool(members.any()):
                centroids[j] = F.normalize(points[members].mean(0), dim=0)
    return centroids


class Fingerprint:
    """k sub-centroids of one expert, rebuilt when its signature has seen ``refresh`` more stream points than at the last build."""

    def __init__(self, centroids: Tensor, built_seen: int) -> None:
        self.centroids, self.built_seen = centroids, int(built_seen)

    @classmethod
    def build(cls, sig, k: int, seed: int) -> "Fingerprint":
        return cls(spherical_kmeans(sig.protos, k, seed=seed), sig.seen)

    def stale(self, sig, refresh: int) -> bool:
        return sig.seen - self.built_seen >= refresh

    def padded(self, k: int) -> Tensor:
        """Exactly k rows (repeating rows when the expert has fewer), so all experts stack into one (E, k, d) tensor."""
        c = self.centroids
        if c.shape[0] >= k:
            return c[:k]
        return c.repeat(-(-k // c.shape[0]), 1)[:k]
```

```python
    def _gate_matrices(self, x: Tensor, eids: List[int], phase: str = "predict"):
        """(B, K) conformal p-values and log-densities. With ``gate_shortlist > 0`` and enough experts the cascade of ``_gate_cascade`` is used; its
        novelty decisions equal the exact gate's (see that method). ``phase`` ('observe' or 'predict') only decides which counters are advanced."""
        stats = self.gate_stats[phase]
        stats["samples"] += x.shape[0]
        if self._cascade_ok(eids):
            return self._gate_cascade(x, eids, stats)
        stats["stage2"] += x.shape[0] * sum(self.experts[e].meta["sig"].n for e in eids) * self.cfg.d
        return self._gate_exact(x, eids)

    def _cascade_ok(self, eids: List[int]) -> bool:
        """The shortlist needs the conformal gate, vMF signatures with at least two prototypes, and enough experts to be worth it."""
        cfg = self.cfg
        if cfg.gate_shortlist <= 0 or cfg.no_conformal or len(eids) < max(cfg.gate_shortlist_min_experts, cfg.gate_shortlist + 1):
            return False
        return all(isinstance(self.experts[e].meta["sig"], VmfSignature) and self.experts[e].meta["sig"].n >= 2 for e in eids)

    def _fingerprints(self, eids: List[int]) -> Tensor:
        """(E, k, d) unit sub-centroids, one block per expert; an expert's block is rebuilt when its signature has seen ``gate_fingerprint_refresh`` more points."""
        k, refresh = self.cfg.gate_fingerprint_k, self.cfg.gate_fingerprint_refresh
        blocks = []
        for e in eids:
            rec = self.experts[e]
            fp = rec.meta.get("fp")
            if fp is None or fp.stale(rec.meta["sig"], refresh):
                fp = Fingerprint.build(rec.meta["sig"], k, seed=self.cfg.seed * 1000 + e)
                rec.meta["fp"] = fp
            blocks.append(fp.padded(k))
        return torch.stack(blocks)

    def _gate_cascade(self, x: Tensor, eids: List[int], stats: dict):
        """Two-stage gate. (1) Stage 1 scores x against every expert's sub-centroids (best cosine). (2) The top ``gate_shortlist`` experts per input
        (plus every infant) get the exact score. Pairs outside the shortlist stay P = 0, LogF = -inf. (3) Second look: every input whose restricted
        p_nov is at most q gets the exact scan over ALL experts.

        Guarantee: the Benjamini-Hochberg novelty flags equal the exact gate's. BH at level q rejects using only p-values <= q (a p above q can never
        satisfy p_(k) <= kq/B); restricted p <= exact p, and every input whose restricted p <= q is recomputed exactly, so the entries <= q coincide and all
        others exceed q in both versions. Only the choice among accepting experts is approximate."""
        cfg, d = self.cfg, self.cfg.d
        b, n_exp = x.shape[0], len(eids)
        cents = self._fingerprints(eids)                                                  # (E, k, d)
        k = cents.shape[1]
        stage1 = (x @ cents.reshape(n_exp * k, d).T).reshape(b, n_exp, k).amax(dim=2)     # (B, E)
        stats["stage1"] += b * n_exp * k * d
        need = torch.zeros(b, n_exp, dtype=torch.bool)
        need.scatter_(1, stage1.topk(cfg.gate_shortlist, dim=1).indices, True)
        for c, e in enumerate(eids):
            if self.experts[e].state == "infant":                                         # infants are few and small and must attract their regime
                need[:, c] = True
        P = torch.zeros(b, n_exp)
        LogF = torch.full((b, n_exp), -math.inf)
        for c, e in enumerate(eids):
            rows = need[:, c].nonzero().squeeze(1)
            if rows.numel():
                P[rows, c], LogF[rows, c] = self._gate_column(e, x[rows])
                stats["stage2"] += int(rows.numel()) * self.experts[e].meta["sig"].n * d
        look = (P.max(dim=1).values <= cfg.q * 1.001).nonzero().squeeze(1)              # the 1.001 only enlarges the set (safe side)
        if look.numel():
            P[look], LogF[look] = self._gate_exact(x[look], eids)
            stats["second_look"] += int(look.numel()) * sum(self.experts[e].meta["sig"].n for e in eids) * d
        return P, LogF
```

```python
def measured_gate_macs_per_input(pool, phase: str = "predict") -> float:
    """Gate multiply-adds per input that ``pool`` actually spent in ``phase`` ('observe' or 'predict'), from its own counters.
```

```python
# ------------------------------------------------------------------------------------------------ RV2-B verdicts (stage 2: fingerprint shortlist)
# research/growmoe_results/rv2b/PREREG.md: does the shortlist cut the gate cost without costing accuracy?
BARS_RV2B = {"C1_mean": -1.0, "C1_lo": -2.0, "C2": 0.5, "C3": {"F-30": 0.20, "S2-30": 0.25, "F-15": 0.25}, "C4_macs": 0.05, "C4_time": 0.5}
EXPECTED_RV2B = {k: "PASS" for k in ("C1", "C2", "C3", "C4", "G")}
_C1_STREAMS = ("F-15", "F-30", "S2-30", "OV-0.3")
_G_STREAMS = ("S1", "S2-5")


def _rules_rv2b(bars: dict, cand: str, ref: str) -> dict:
    return {
        "C1": f"F-15, F-30, S2-30, OV-0.3: mean paired {cand} - {ref} >= {bars['C1_mean']:+.1f} points and the interval lower bound >= {bars['C1_lo']:+.1f}",
        "C2": f"the mean absolute difference in spawns per run between {cand} and {ref} is <= {bars['C2']} on those streams",
        "C3": "measured gate multiply-adds per input, " + ", ".join(f"{cand}/{ref} <= {v:.2f} on {s}" for s, v in bars["C3"].items()),
        "C4": f"synthetic pool of 500 experts: gate MACs ratio <= {bars['C4_macs']} and gate wall-clock ratio <= {bars['C4_time']} at B = 64 and B = 1024",
        "G": "S1 and S2-5 (too few experts for the cascade): accuracy and expert count bit-identical to the exact gate",
    }


def rv2b_verdicts(runs: Sequence[dict], scaling: dict = None, bars: dict = None, expected: dict = None,
                  candidate: str = "A_cascade", reference: str = "A_exact") -> dict:
    bars = {**BARS_RV2B, **(bars or {})}
    exp = {**EXPECTED_RV2B, **(expected or {})}
    ix = _index(runs)
    c1, c2, c3, g = {}, {}, {}, {}
    for s in _C1_STREAMS:
        d = _paired_points(ix, (s, candidate), (s, reference), bars["C1_mean"])
        if d is not None:
            d["holds"] = bool(d["holds"] and d["ci_lo"] is not None and d["ci_lo"] >= bars["C1_lo"])
        c1[s] = d
        pair = _common(ix, (s, candidate), (s, reference))
        if pair is None:
            c2[s] = None
        else:
            a, b, seeds = pair
            gap = float(np.mean([abs(a[i]["event_counts"].get("spawn", 0) - b[i]["event_counts"].get("spawn", 0)) for i in seeds]))
            c2[s] = {"n": len(seeds), "mean_abs_spawn_gap": gap, "holds": gap <= bars["C2"]}
    for s, bar in bars["C3"].items():
        pair = _common(ix, (s, candidate), (s, reference))
        if pair is None:
            c3[s] = None
        else:
            a, b, seeds = pair
            ratio = float(np.mean([a[i]["gate_macs_pred"] for i in seeds]) / np.mean([b[i]["gate_macs_pred"] for i in seeds]))
            c3[s] = {"n": len(seeds), "ratio": ratio, "bar": bar, "holds": ratio <= bar}
    for s in _G_STREAMS:
        pair = _common(ix, (s, candidate), (s, reference))
        if pair is None:
            g[s] = None
        else:
            a, b, seeds = pair
            same = all(a[i]["acc"] == b[i]["acc"] and a[i]["n_experts"] == b[i]["n_experts"] for i in seeds)
            g[s] = {"n": len(seeds), "identical": same, "holds": bool(same)}
    detail = {"C1": c1, "C2": c2, "C3": c3, "G": g}
    row = (scaling or {}).get("rows", {}).get("500")
    if not row or not all(b in row for b in ("64", "1024")):
        detail["C4"] = None
    else:
        ok = all(row[b]["ratio_macs"] <= bars["C4_macs"] and row[b]["ratio_time"] <= bars["C4_time"] for b in ("64", "1024"))
        detail["C4"] = {"holds": bool(ok), **{f"B{b}": {"ratio_macs": row[b]["ratio_macs"], "ratio_time": row[b]["ratio_time"]} for b in ("64", "1024")}}
    verdict = {k: _combine(v) for k, v in detail.items()}
    rules = _rules_rv2b(bars, candidate, reference)
    return {k: {"rule": rules[k], "verdict": verdict[k], "detail": detail[k], "expected": exp[k],
                "as_expected": None if verdict[k] == "NOT MEASURABLE" else verdict[k] == exp[k]} for k in EXPECTED_RV2B}


def rv2b_decision(result: dict, k: int = None, min_experts: int = None) -> str:
    """What the pre-registration says to do. C1 is the correctness statement; C3 and C4 decide default vs opt-in; G failing is a bug."""
    if result["C1"]["verdict"] == "NOT MEASURABLE":
        return "The RV2-B decision cannot be applied: C1 was not measurable."
    if result["G"]["verdict"] == "FAIL":
        return "G FAIL: a pool too small for the cascade does not match the exact gate bit for bit. Stop: this is a bug, not a result."
    if result["C1"]["verdict"] != "PASS":
        return f"C1 {result['C1']['verdict']}: the cascade is not shown to be non-inferior; it is not adopted and gate_shortlist stays 0."
    if result["C3"]["verdict"] == "PASS" and result["C4"]["verdict"] == "PASS":
        return (f"C1, C3 and C4 PASS: adopt the cascade as the PhyloMoE default (gate_shortlist=2, gate_fingerprint_k={k}, "
                f"gate_shortlist_min_experts={min_experts}), test-first, with the historical entry points pinned.")
    return ("C1 PASS but the cost statements C3/C4 did not both pass: keep the cascade opt-in (gate_shortlist stays 0 by default) and say so "
            "wherever it is documented.")
```

```python
def run_task(spec: dict) -> dict:
    name, seed, smoke = spec["stream"], spec["seed"], spec["smoke"]
    st = A.get_stream(name, seed, smoke)
    steps, t0, recs = steps_for(name), time.perf_counter(), []
    for arm in spec["arms"]:
        k = int(arm.split("_k")[1]) if "_k" in arm else spec["k"]
        holder = []

        def make(arm=arm, k=k):
            holder.append(factory(arm.split("_k")[0], st, seed, steps, k)())
            return holder[-1]

        res = run_fast(make, st, seed, eval_every="final")
        pool = holder[0]
        recs.append(run_record(name, arm, seed, res, steps=steps, k=k, gate_macs_pred=measured_gate_macs_per_input(pool, "predict"),
                               gate_macs_obs=measured_gate_macs_per_input(pool, "observe")))
    return {"spec": {k_: spec[k_] for k_ in ("stream", "seed", "smoke", "arms")}, "records": recs, "seconds": time.perf_counter() - t0, "peak_mb": M._peak_mb()}
```

```python
"""THROWAWAY spike on DEV seeds (100-102): a CALIBRATED batch-level novelty test for regimes that merge at high overlap (spec section 6).

After the stage-1 vote, take the expert that most of the batch was routed to. Compare the batch with that expert's own regime: T = ||mean(batch) -
mean of the expert's even-indexed prototypes|| x sqrt(b). The null is the same statistic on bootstrap batches drawn from the expert's ODD-indexed
prototypes (a cross-fit, so the null is not tied to the data that defined the mean), threshold = its (1 - alpha) quantile x margin. A flagged batch
goes whole to the probation buffer as a candidate cluster (patched copy of PhyloMoE.observe; no product file touched).

Acceptance (spec section 6), on dev seeds, against stage-1-only (batch_vote=True):
  (a) expert counts unchanged on F-15, F-30, S1, S2-5, S2-30, OV-0.0;
  (b) at OV-0.6 and OV-0.85: at least 8 experts for 10 regimes and at least +8 points accuracy.

    python research/growmoe_results/rv2/spike_batch_novelty.py
"""

import inspect
import json
import math
import os
import sys
import textwrap
from pathlib import Path

HERE = Path(__file__).resolve().parent
for _p in (HERE.parent / "rv2a", HERE.parent / "m2a", HERE.parent / "m0", HERE.parents[2]):
    sys.path.insert(0, str(_p))
os.environ.setdefault("OMP_NUM_THREADS", "3")

import numpy as np                                                                      # noqa: E402
import torch                                                                            # noqa: E402

import run_rv2a as A                                                                    # noqa: E402
from growmoe.config import PhyloCfg                                                     # noqa: E402
from growmoe.pools import phylo as phylo_mod                                            # noqa: E402
from growmoe.pools.phylo import PhyloMoE                                                # noqa: E402

torch.set_num_threads(3)

_PATCH = """\
if getattr(self, "batch_novelty", False):
    _ok = assigned >= 0
    if int(_ok.sum()) >= self.cfg.batch_vote_min:
        _vals, _cnts = torch.unique(assigned[_ok], return_counts=True)
        _protos = self.experts[int(_vals[int(_cnts.argmax())])].meta["sig"].protos
        if _protos.shape[0] >= 16 and self._batch_is_foreign(x[_ok], _protos):
            novel = torch.ones_like(novel)
            assigned = torch.full_like(assigned, -1)
"""


def _patched_observe():
    lines = textwrap.dedent(inspect.getsource(PhyloMoE.observe)).split("\n")
    i = next(k for k, l in enumerate(lines) if l.strip() == "spawned = []")
    indent = lines[i][: len(lines[i]) - len(lines[i].lstrip())]
    lines[i:i] = [indent + l if l else l for l in _PATCH.rstrip("\n").split("\n")]
    ns: dict = {}
    exec("\n".join(lines), vars(phylo_mod), ns)
    return ns["observe"]


class NoveltyPool(PhyloMoE):
    batch_novelty = False
    alpha, margin, n_null = 1e-3, 1.2, 400

    def _batch_is_foreign(self, xb, protos) -> bool:
        b = xb.shape[0]
        mu = protos[0::2].mean(0)
        null_pool = protos[1::2]
        g = torch.Generator().manual_seed(int(self.samples_seen))
        idx = torch.randint(0, null_pool.shape[0], (self.n_null, b), generator=g)
        null = (null_pool[idx].mean(1) - mu).norm(dim=1) * math.sqrt(b)
        thr = float(torch.quantile(null, 1.0 - self.alpha)) * self.margin
        return float((xb.mean(0) - mu).norm() * math.sqrt(b)) > thr


NoveltyPool.observe = _patched_observe()
STREAMS = ["F-15", "F-30", "S1", "S2-5", "S2-30", "OV-0.0", "OV-0.6", "OV-0.85"]
SEEDS = (100, 101, 102)


def run(name, seed, novelty):
    st = A.get_stream(name, seed, False)
    tm_new, tm_vote = A.load_tunings() if hasattr(A, "load_tunings") else (None, None)
    steps = 5 if name.startswith(("OV-", "F-")) else json.loads(Path(A.M2A_TUNING).read_text(encoding="utf-8"))["A_new"][A.family(name)]["steps"]
    pool = NoveltyPool(PhyloCfg(seed=seed, d=st.d, n_classes=st.n_classes, inner_steps=steps, **{**A.M.A_KW, "arm_monitor_after_infancy": True, "batch_vote": True}))
    pool.batch_novelty = novelty
    for x, y, ctx in st.algo_batches(64, seed):
        pool.observe(x, y, ctx)
    x = torch.cat([r.x_te for r in st.regimes.values()])
    y = torch.cat([r.y_te for r in st.regimes.values()])
    return {"stream": name, "seed": seed, "novelty": novelty, "acc": float((pool.predict(x).probs.argmax(1) == y).float().mean()), "experts": len(pool.experts)}


if __name__ == "__main__":
    rows = []
    for name in STREAMS:
        for novelty in (False, True):
            for seed in SEEDS:
                rows.append(run(name, seed, novelty))
                print(rows[-1], flush=True)
    (HERE / "spike_batch_novelty.json").write_text(json.dumps(rows, indent=1), encoding="utf-8")
    m = lambda s, nv, k: float(np.mean([r[k] for r in rows if r["stream"] == s and r["novelty"] == nv]))     # noqa: E731
    print("\nstream     experts: vote-only -> +novelty     accuracy %: vote-only -> +novelty")
    for s in STREAMS:
        print(f"{s:9s} {m(s, False, 'experts'):8.1f} -> {m(s, True, 'experts'):5.1f}        {100 * m(s, False, 'acc'):6.1f} -> {100 * m(s, True, 'acc'):5.1f}")
    a_ok = all(abs(m(s, True, "experts") - m(s, False, "experts")) < 0.5 for s in ("F-15", "F-30", "S1", "S2-5", "S2-30", "OV-0.0"))
    b_ok = all(m(s, True, "experts") >= 8 and 100 * (m(s, True, "acc") - m(s, False, "acc")) >= 8.0 for s in ("OV-0.6", "OV-0.85"))
    print(f"\n(a) no spurious experts: {'MET' if a_ok else 'NOT MET'}   (b) merges cured at OV-0.6 and OV-0.85: {'MET' if b_ok else 'NOT MET'}")
```
